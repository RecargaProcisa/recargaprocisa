AUTO-PREENCHIMENTO DO NOME VIA MATRÍCULA (À Prova de Falhas) const inputMatricula = document.getElementById('matricula'); const campoNome = document.getElementById('nome_solicitante'); async function buscarNomeColaborador() { const mat = inputMatricula.value.trim(); if (mat.length > 0) { campoNome.placeholder = "A pesquisar..."; campoNome.value = ""; // Limpa o campo para nova busca try { // Usamos .ilike para ignorar diferenças de formatação e removemos o .single() const { data, error } = await supabaseClient .from('base_colaboradores') .select('nome') .ilike('matricula', mat) .limit(1); if (error) throw error; if (data && data.length > 0) { campoNome.value = data[0].nome; campoNome.classList.add('bg-emerald-50', 'border-emerald-400', 'text-emerald-900', 'font-bold'); setTimeout(() => campoNome.classList.remove('bg-emerald-50', 'border-emerald-400', 'text-emerald-900', 'font-bold'), 3000); } else { campoNome.placeholder = "NOME NÃO ENCONTRADO. DIGITE MANUALMENTE."; } } catch (err) { console.error("Erro na busca do Supabase:", err); campoNome.placeholder = "ERRO DE COMUNICAÇÃO. DIGITE MANUALMENTE."; } } } // Aciona a busca tanto ao sair do campo (blur) quanto ao digitar (change/keyup) inputMatricula.addEventListener('blur', buscarNomeColaborador); inputMatricula.addEventListener('keyup', (e) => { if (e.key === 'Enter') buscarNomeColaborador(); });
